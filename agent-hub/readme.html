一、架构概览

1.1 项目架构定位

AgentHub 是一个典型的分层架构项目，采用微服务前期的模块化单体设计。这种架构既保证了开发效率，又为未来的微服务拆分预留了空间。

自定义中间件实现

use axum::{
    extract::Request,
    middleware::Next,
    response::Response,
};

async fn request_id_middleware(
    req: Request,
    next: Next,
) -> Response {
    // 生成请求 ID
    let request_id = Uuid::new_v4().to_string();

    // 添加到请求扩展
    let mut req = req;
    req.extensions_mut().insert(request_id.clone());

    // 调用下一个中间件
    let mut response = next.run(req).await;

    // 添加到响应头
    response.headers_mut().insert(
        "x-request-id",
        request_id.parse().unwrap(),
    );

    response
}

macro_rules! define_id {
    ($name:ident) => {
        #[derive(Debug, Clone, PartialEq, Eq, Hash, Serialize, Deserialize)]
        #[serde(transparent)]
        pub struct $name(String);

        impl $name {
            pub fn generate() -> Self {

            pub fn as_str(&self) -> &str {
                &self.0
            }
        }

        impl From for $name {
            fn from(s: String) -> Self {
                Self(s)
            }
        }
    };
}

内存实现示例

// 内存存储实现
use dashmap::DashMap;
use std::sync::Arc;

pub struct InMemoryRegistry {
    agents: Arc>,
    capabilities: Arc>>,
}

impl InMemoryRegistry {
    pub fn new() -> Self {
        Self {
            agents: Arc::new(DashMap::new()),
            capabilities: Arc::new(DashMap::new()),
        }
    }
}

#[async_trait]
impl AgentRegistry for InMemoryRegistry {
    async fn register(&self, agent: AgentDescriptor) -> Result {
        let agent_id = agent.id.clone();

        // 存储 agent
        self.agents.insert(agent_id.clone(), agent.clone());

        // 更新能力索引
        for capability in &agent.capabilities {
            self.capabilities
                .entry(capability.clone())
                .or_insert_with(Vec::new)
                .push(agent_id.clone());
        }

        Ok(agent_id)
    }

    async fn discover(&self, capability: &str) -> Result> {
        let agent_ids = self.capabilities.get(capability)
            .map(|ids| ids.clone())
            .unwrap_or_default();

        let agents: Vec = agent_ids
            .iter()
            .filter_map(|id| self.agents.get(id).map(|r| r.clone()))
            .collect();

        Ok(agents)
    }
}

PostgreSQL 实现

// PostgreSQL Repository 实现
pub struct PostgresAgentRepository {
    pool: PgPool,
}

impl PostgresAgentRepository {
    pub fn new(pool: PgPool) -> Self {
        Self { pool }
    }
}

#[async_trait]
impl AgentRepository for PostgresAgentRepository {
    async fn save(&self, agent: &AgentDescriptor) -> Result<()> {
        sqlx::query(
            r#"
            INSERT INTO agents (id, name, endpoint, capabilities, status, tenant_id)
            VALUES ($1, $2, $3, $4, $5, $6)
            ON CONFLICT (id) DO UPDATE SET
                name = EXCLUDED.name,
                endpoint = EXCLUDED.endpoint,
                capabilities = EXCLUDED.capabilities,
                status = EXCLUDED.status
            "#
        )
        .bind(&agent.id)
        .bind(&agent.name)
        .bind(&agent.endpoint)
        .bind(&agent.capabilities)
        .bind(&agent.status)
        .bind(&agent.tenant_id)
        .execute(&self.pool)
        .await?;

        Ok(())
    }

    async fn find_by_id(&self, id: &AgentId) -> Result> {
        let agent = sqlx::query_as::<_, AgentDescriptor>(
            "SELECT * FROM agents WHERE id = $1"
        )
        .bind(id)
        .fetch_optional(&self.pool)
        .await?;

        Ok(agent)
    }
}

四、API 接口设计

4.1 RESTful 设计规范

资源命名规范

✅ 好的设计 
GET    /v1/agents              # 获取 Agent 列表
GET    /v1/agents/{id}         # 获取特定 Agent
POST   /v1/agents              # 创建新 Agent
PUT    /v1/agents/{id}         # 完整更新 Agent
PATCH  /v1/agents/{id}         # 部分更新 Agent
DELETE /v1/agents/{id}         # 删除 Agent
POST   /v1/agents/{id}/heartbeat  # Agent 心跳
❌ 不好的设计 
GET    /v1/getAgents           # 动词在 URL 中
POST   /v1/agents/create       # 重复表达
GET    /v1/agents?id=123       # 用查询参数表示资源 ID
POST   /v1/agents/register     # 动词作为资源名

HTTP 方法语义

方法语义幂等性缓存性AgentHub 用途
GET获取资源是可缓存查询 Agent、获取配置
POST创建资源否不可缓存注册 Agent、发起调用
PUT完整更新是不可缓存更新 Agent 配置
PATCH部分更新否不可缓存更新 Agent 状态
DELETE删除资源是不可缓存注销 Agent

4.4 流式响应实现

// SSE 流式响应
use axum::{
    response::{sse::{Event, Sse},
    body::Body,
};
use futures::stream::Stream;
use std::convert::Infallible;

async fn stream_invoke(
    State(deps): State,
    Json(req): Json,
) -> Sse>> {
    let (sender, receiver) = tokio::sync::mpsc::channel(100);

    // 启动后台任务生成流式数据
    tokio::spawn(async move {
        let agent = deps.registry.discover(&req.model).await.unwrap();
        let client = deps.agent_client.get_client(&agent.endpoint).await.unwrap();

        match client.invoke_stream(req).await {
            Ok(mut stream) => {
                while let Some(chunk) = stream.next().await {
                    let event = Event::default()
                        .json_data(chunk)
                        .unwrap();

                    sender.send(Ok(event)).await.unwrap();
                }
            }
            Err(e) => {
                let error_event = Event::default()
                    .json_data(serde_json::json!({"error": e.to_string()}))
                    .unwrap();
                sender.send(Ok(error_event)).await.unwrap();
            }
        }
    });

    Sse::new(receiver).keep_alive(
        axum::response::sse::KeepAlive::new()
            .interval(Duration::from_secs(10))
            .text("keepalive"),
    )
}

五、数据流转机制

5.1 请求处理完整流程

从 HTTP 请求到响应的完整数据流

1. TCP 连接建立
   Client → TCP 三次握手 → Server
   Client → TLS 握手 (HTTPS) → Server
   ↓
2. HTTP 请求解析
   Axum 接收 TCP Stream
   解析 HTTP Request Line: POST /v1/agents HTTP/1.1
   解析 Headers: Content-Type, Authorization, etc.
   解析 Body: JSON 数据
   ↓
3. 中间件链处理
   ┌─────────────────────────────────┐
   │ Request ID Middleware           │ ← 生成/提取请求 ID
   │ Auth Middleware                │ ← 验证 JWT/API Key
   │ Rate Limit Middleware           │ ← 检查请求频率
   │ Logging Middleware              │ ← 记录请求日志
   │ Timeout Middleware              │ ← 设置超时控制
   └─────────────────────────────────┘
   ↓
4. 路由匹配

5.2 中间件数据传递

// 使用 Extensions 在中间件间传递数据
async fn auth_middleware(
    mut req: Request,
    next: Next,
) -> Result {
    // 提取认证信息
    let auth_header = req.headers()
        .get("Authorization")
        .and_then(|h| h.to_str().ok());

    if let Some(token) = auth_header {
        // 验证令牌
        if let Ok(principal) = verify_token(token) {
            // 将认证信息存入请求扩展
            req.extensions_mut().insert(principal);
            return Ok(next.run(req).await);
        }
    }

    Err(StatusCode::UNAUTHORIZED)
}

// 在 handler 中提取认证信息
async fn protected_handler(
    req: Request,
) -> Result, StatusCode> {
    // 从请求扩展中提取认证信息
    let principal = req.extensions()
        .get::()
        .ok_or(StatusCode::UNAUTHORIZED)?;

    // 使用认证信息
    Ok(Json(Response {
        user_id: principal.user_id.clone(),
        tenant_id: principal.tenant_id.clone(),
    }))
}

            BusinessError::Validation(msg) => AppError::BadRequest(msg),

// 3. 应用错误转换为 HTTP 响应
impl IntoResponse for AppError {
    fn into_response(self) -> Response {
        let (status, code, message) = match self {
            AppError::NotFound(msg) => (StatusCode::NOT_FOUND, 40401, msg),
            AppError::BadRequest(msg) => (StatusCode::BAD_REQUEST, 40001, msg),
            AppError::Internal(msg) => (StatusCode::INTERNAL_SERVER_ERROR, 50001, msg),
        };

        let body = Json(json!({
            "code": code,
            "message": message,
            "timestamp": Utc::now().timestamp()
        }));

        (status, body).into_response()
    }
}

7.4 限流实现

// 基于 token bucket 的限流
use std::sync::Arc;
use std::time::{Duration, Instant};
use tokio::sync::Mutex;

struct RateLimiter {
    capacity: u32,
    tokens: Arc>,
    last_refill: Arc>,
    refill_rate: u32, // tokens per second
}

impl RateLimiter {
    fn new(capacity: u32, refill_rate: u32) -> Self {
        Self {
            capacity,
            tokens: Arc::new(Mutex::new(capacity)),
            last_refill: Arc::new(Mutex::new(Instant::now())),
            refill_rate,
        }
    }

    async fn acquire(&self) -> bool {
        let mut tokens = self.tokens.lock().await;
        let mut last_refill = self.last_refill.lock().await;

        // 补充 token
        let now = Instant::now();
        let elapsed = now.duration_since(*last_refill).as_secs() as u32;
        let new_tokens = (elapsed * self.refill_rate).min(self.capacity);

        *tokens = (*tokens + new_tokens).min(self.capacity);
        *last_refill = now;

        // 检查是否有足够 token
        if *tokens > 0 {
            *tokens -= 1;
            true
        } else {
            false
        }
    }
}

8.2 缓存策略

多级缓存

// 三级缓存：内存 → Redis → 数据库
use std::collections::HashMap;
use std::sync::Arc;
use tokio::sync::RwLock;

struct MultiLevelCache {
    l1: Arc>>,  // 内存缓存
    l2: Arc,                      // Redis 缓存
    l3: Arc,                           // 数据库
}

impl MultiLevelCache {
    async fn get(&self, key: &str) -> Option {
        // L1: 内存缓存
        {
            let cache = self.l1.read().await;
            if let Some(value) = cache.get(key) {
                return Some(value.clone());
            }
        }

        // L2: Redis 缓存
        if let Ok(Some(value)) = self.l2.get(key).await {
            // 回填 L1
            let mut cache = self.l1.write().await;
            cache.insert(key.to_string(), value.clone());
            return Some(value);
        }

        // L3: 数据库
        if let Ok(Some(value)) = self.fetch_from_db(key).await {
            // 回填 L2 和 L1
            self.l2.set_ex(key, &value, 300).await.ok();
            let mut cache = self.l1.write().await;
            cache.insert(key.to_string(), value.clone());
            return Some(value);
        }

        None
    }
}
